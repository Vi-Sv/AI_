## Блок 5: Главный оркестратор и системная служба перехвата событий Windows
В этом финальном блоке мы разработаем интеллектуальное ядро (Оркестратор), которое свяжет воедино память, векторную базу знаний RAG и клиент нейросети. Затем мы напишем главную точку входа системы — службу, которая работает в фоновом режиме Windows, регистрирует глобальные системные хуки (горячие клавиши) и реализует продвинутый маневр: автоматический анализ выделенного на экране текста.
------------------------------
## 1. Разработка главного оркестратора (core/agent.py)
Этот компонент управляет жизненным циклом одного запроса. Он принимает текст, обращается к ChromaDB за контекстом документов, извлекает историю переписки, собирает сложный системный промпт (Prompt Engineering) и передает готовую структуру в Ollama.
Создайте файл core/agent.py и добавьте следующий код:

import osfrom core.llm_client import OllamaClientfrom core.memory import ChatMemoryfrom rag.vector_store import VectorStoreManager
class AIAgent:
    def __init__(self):
        """Главный оркестратор ИИ-агента, связывающий UI, LLM, RAG и память."""
        self.client = OllamaClient()
        self.memory = ChatMemory(max_turns=10)
        self.rag = VectorStoreManager()
        
        # Системные инструкции, определяющие характер и правила поведения ИИ
        self.system_prompt = (
            "Ты — продвинутый локальный ИИ-ассистент, глубоко интегрированный в операционную систему Windows.\n"
            "Твоя задача — давать точные, технически выверенные и лаконичные ответы на русском языке.\n"
            "Если тебе предоставлен локальный контекст из документов, приоритетно используй его для ответа.\n"
            "Отвечай строго по делу, без пространных вступлений и лишней вежливости."
        )

    def stream_query(self, user_query: str):
        """
        Полный цикл обработки запроса с потоковой передачей ответа в UI.
        1. Поиск релевантных фактов в локальной базе знаний ChromaDB (RAG).
        2. Формирование обогащенного промпта.
        3. Добавление запроса в контекст памяти.
        4. Стриминг чанков обратно в поток графического интерфейса.
        """
        # Шаг 1: Извлекаем контекст из документов
        rag_context = self.rag.query_knowledge_base(user_query)
        
        # Шаг 2: Формируем финальный промпт для модели
        if rag_context:
            full_prompt = (
                f"Используй следующие документы для ответа на вопрос пользователя.\n"
                f"{rag_context}\n"
                f"Вопрос пользователя: {user_query}"
            )
        else:
            full_prompt = user_query

        # Шаг 3: Записываем реплику пользователя в память сессии
        self.memory.add_message(role="user", content=full_prompt)
        
        # Переменная для сборки полного ответа для последующего логирования
        full_ai_response = ""
        
        # Шаг 4: Запрашиваем потоковый ответ у Ollama
        try:
            for chunk in self.client.stream_response(
                history=self.memory.get_context(), 
                system_prompt=self.system_prompt
            ):
                full_ai_response += chunk
                yield chunk  # Передаем чанк в интерфейс в реальном времени
                
            # После успешного завершения генерации сохраняем ответ в память и на диск
            if full_ai_response:
                self.memory.add_message(role="assistant", content=full_ai_response)
                self.memory.save_to_disk(user_query, full_ai_response)
                
        except Exception as e:
            yield f"\n[Критическая ошибка оркестратора]: {str(e)}"

## 2. Разработка службы автоматизации Windows (main.py)
Этот скрипт является точкой входа (Entry Point). Он запускает графический интерфейс в скрытом режиме, вешает низкоуровневые хуки на клавиатуру Windows и реализует функцию продвинутого анализа выделенного текста: при нажатии Ctrl+Alt+X скрипт имитирует нажатие Ctrl+C, вытаскивает текст из буфера обмена оперативной памяти, мгновенно разворачивает окно чата и отправляет этот текст ИИ с командой «Объясни/исправь».
Создайте файл main.py в корневой директории проекта:

import osimport sysimport timeimport timeitimport keyboardimport pyperclip
# Добавляем текущую директорию в пути для исключения ошибок импорта внутри пакетов
sys.path.append(os.path.dirname(os.path.abspath(__file__)))
from core.agent import AIAgentfrom ui.chat_window import ChatWindow
class WindowsAIService:
    def __init__(self):
        """Фоновая служба Windows для управления ИИ-агентом через глобальные хоткеи."""
        print("[INIT] Инициализация ядра ИИ-агента...")
        self.agent = AIAgent()
        
        print("[INIT] Сборка графического интерфейса...")
        self.ui = ChatWindow(self.agent)
        
        # По умолчанию скрываем окно, чтобы приложение работало незаметно в фоне
        self.ui.withdraw()
        
        # Регистрация глобальных горячих клавиш Windows
        self.register_hotkeys()

    def register_hotkeys(self):
        """Безопасная привязка методов к системным комбинациям клавиш."""
        # Хоткей 1: Просто вызвать или скрыть окно чата (Ctrl + Alt + Z)
        keyboard.add_hotkey("ctrl+alt+z", self.toggle_chat_window)
        
        # Хоткей 2: Прочесть выделенный на экране текст и отправить ИИ (Ctrl + Alt + X)
        keyboard.add_hotkey("ctrl+alt+x", self.analyze_selected_text)
        
        print("[SUCCESS] Глобальные хоткеи зарегистрированы:")
        print("          - Ctrl + Alt + Z : Вызов/скрытие окна чата")
        print("          - Ctrl + Alt + X : Анализ выделенного текста/кода")

    def toggle_chat_window(self):
        """Переключает видимость окна чата."""
        if self.ui.winfo_viewable():
            self.ui.hide_window()
        else:
            self.ui.show_window()

    def analyze_selected_text(self):
        """
        Продвинутый маневр: захват выделенного текста через буфер обмена Windows.
        """
        # Сохраняем текущее содержимое буфера пользователя, чтобы не затереть его личные данные
        old_clipboard = pyperclip.paste()
        
        # Очищаем буфер перед операцией
        pyperclip.copy("")
        
        # Эмулируем аппаратное нажатие Ctrl+C для копирования выделенного текста в фокусе ОС
        keyboard.send("ctrl+c")
        
        # Ждем короткую паузу, пока Windows обработает команду копирования в ОЗУ
        time.sleep(0.15)
        
        # Считываем скопированный текст
        selected_text = pyperclip.paste().strip()
        
        # Возвращаем пользователю его исходный буфер обмена
        pyperclip.copy(old_clipboard)
        
        if not selected_text:
            print("[WARNING] Не удалось захватить текст. Возможно, ничего не было выделено.")
            return

        # Разворачиваем окно чата на передний план
        self.ui.show_window()
        
        # Формируем контекстный запрос для ИИ
        prompt_instruction = f"Проанализируй, исправь ошибки или объясни следующий фрагмент текста/кода:\n\n{selected_text}"
        
        # Эмулируем отправку сообщения, как будто пользователь ввел его в GUI
        self.ui.append_to_chat("Вы", f"[Анализ выделенного текста]: {selected_text[:100]}...")
        self.ui.append_to_chat("ИИ", "")
        
        # Запускаем генерацию в фоновом потоке интерфейса
        import threading
        threading.Thread(
            target=self.ui.async_ai_worker, 
            args=(prompt_instruction,), 
            daemon=True
        ).start()

    def run(self):
        """Запуск главного бесконечного цикла обработки событий UI."""
        print("[RUN] Фоновая служба запущена. Ожидание команд...")
        self.ui.mainloop()
if __name__ == "__main__":
    # Гарантируем наличие необходимых служебных директорий перед стартом
    os.makedirs("data/documents", exist_ok=True)
    os.makedirs("data/db", exist_ok=True)
    
    service = WindowsAIService()
    service.run()

## 3. Скрипт автоматизации запуска (run.bat)
Чтобы запускать вашу систему в один клик без необходимости открывать терминал и вручную активировать виртуальное окружение, создайте в корне проекта файл run.bat:

@echo off
cd /d "%~dp0"
title Local AI Agent Service
echo Запуск локального ИИ-агента в фоновом режиме...
start "" "venv\Scripts\pythonw.exe" main.py
exit

(Использование pythonw.exe вместо python.exe полностью скрывает черное окно консоли Windows, оставляя процесс работать исключительно в оперативной памяти).
